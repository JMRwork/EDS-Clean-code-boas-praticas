package br.edu.infnet.appFrete.service;

import java.math.BigDecimal;

import br.edu.infnet.appFrete.model.Entrega;
import br.edu.infnet.appFrete.service.calculadora.CalculadoraFrete;

public class EtiquetaService {
	private CalculadoraFrete calculadoraFrete;

	public EtiquetaService(CalculadoraFrete calculadora) {
		this.calculadoraFrete = calculadora;
	}

	public String gerarEtiqueta(Entrega entrega) {
		if (PromocaoService.isFreteGratis(entrega)) {
			return String.format("""
					Destinatário: %s
					Endereço: %s
					Tipo de Frete: %s
					Valor do Frete: GRÁTIS
					""", entrega.getDestinatario(), entrega.getEndereco(), entrega.getTipoFrete().getCodigo());
		}

		BigDecimal valorFrete = calculadoraFrete.calcular(PromocaoService.getPesoComDesconto(entrega.getPeso()));
		return String.format("""
				Destinatário: %s
				Endereço: %s
				Tipo de Frete: %s
				Valor do Frete: R$%.2f
				""", entrega.getDestinatario(), entrega.getEndereco(), entrega.getTipoFrete().getCodigo(), valorFrete);

	}

	public String gerarResumoPedido(Entrega entrega) {
		if (PromocaoService.isFreteGratis(entrega)) {
			return String.format("Pedido para %s com frete %s GRÁTIS", entrega.getDestinatario(),
					entrega.getTipoFrete().getCodigo());
		}
		
		BigDecimal valorFrete = calculadoraFrete.calcular(PromocaoService.getPesoComDesconto(entrega.getPeso()));
		return String.format("Pedido para %s com frete %s no valor de R$%.2f", entrega.getDestinatario(),
				entrega.getTipoFrete().getCodigo(), valorFrete);
	}

}
